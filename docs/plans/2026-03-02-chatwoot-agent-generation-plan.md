// =============================================================================

export function shouldProcessWebhook(payload: ChatwootWebhookPayload): {
  shouldProcess: boolean;
  reason?: string;
} {
  // Only process message_created events
  if (payload.event !== 'message_created') {
    return { shouldProcess: false, reason: `Event type: ${payload.event}` };
  }

  // Only process incoming messages (from customers)
  if (payload.message_type !== 'incoming') {
    return { shouldProcess: false, reason: `Message type: ${payload.message_type}` };
  }

  // Skip messages from agents or bots
  const senderType = payload.sender?.type;
  if (senderType === 'user' || senderType === 'agent_bot') {
    return { shouldProcess: false, reason: `Sender type: ${senderType}` };
  }

{% if webhooks.chatwoot.get("filter_open_conversations", true) %}
  // Skip conversations that are "open" (human agent is handling)
  const conversationStatus = payload.conversation?.status;
  if (conversationStatus === 'open') {
    return { shouldProcess: false, reason: `Conversation status: ${conversationStatus} (human handling)` };
  }
{% endif %}

  // Check for duplicate messages
  if (processedMessages.has(payload.id)) {
    return { shouldProcess: false, reason: 'Duplicate message' };
  }

  // Mark as processed and schedule cleanup
  processedMessages.add(payload.id);
  setTimeout(() => processedMessages.delete(payload.id), MESSAGE_CACHE_TTL);

  return { shouldProcess: true };
}

  const parts: string[] = [];

  for (const attachment of attachments) {
    if (['audio', 'image', 'video'].includes(attachment.file_type)) continue;

    switch (attachment.file_type) {
      case 'location':
        if (attachment.coordinates_lat && attachment.coordinates_long) {
          parts.push(`[Location shared: Lat ${attachment.coordinates_lat}, Long ${attachment.coordinates_long}]`);
        } else {
          parts.push(`[Location shared: ${attachment.fallback_title || 'no coordinates'}]`);
        }
        break;
      case 'file':
        parts.push(`[File attachment: ${attachment.extension || 'document'}]`);
        break;
      case 'contact':
        parts.push('[Contact shared]');
        break;
      default:
        parts.push(`[Attachment type: ${attachment.file_type}]`);
    }
  }

  return [mediaResult, ...parts].filter(p => p.trim()).join('\n');
}

// =============================================================================
// Download Media
// =============================================================================

async function downloadMedia(
  url: string,
  retries = 3,
): Promise<{ data: Buffer; contentType: string } | null> {
  for (let attempt = 1; attempt <= retries; attempt++) {
    try {
      const response = await fetch(url);
      if (!response.ok) {
        if (response.status === 404 && attempt < retries) {
          await new Promise(r => setTimeout(r, attempt * 2000));
          continue;
        }
        console.error(`[Media] Download failed: ${response.status}`);
        return null;
      }
      const contentType = response.headers.get('content-type') || 'application/octet-stream';
      const arrayBuffer = await response.arrayBuffer();
      return { data: Buffer.from(arrayBuffer), contentType };
    } catch (error) {
      if (attempt < retries) {
        await new Promise(r => setTimeout(r, attempt * 2000));
        continue;
      }
      console.error('[Media] Download error:', error);
      return null;
    }
  }
  return null;
}

async function transcribeAudio(audioBuffer: Buffer, contentType: string): Promise<string> {
  const apiKey = process.env.OPENAI_API_KEY;
  if (!apiKey) {
    return '[Audio message received but transcription is not configured]';
  }

  try {
    const ext = contentType.includes('ogg') ? 'ogg' : contentType.includes('mp4') ? 'mp4' : 'wav';
    const blob = new Blob([audioBuffer], { type: contentType });
    const file = new File([blob], `audio.${ext}`, { type: contentType });

    const formData = new FormData();
    formData.append('file', file);
    formData.append('model', 'whisper-1');
    formData.append('language', 'es');

      method: 'POST',
      headers: { Authorization: `Bearer ${apiKey}` },
      body: formData,

      return '[Audio message could not be transcribed]';

# Track processed message IDs to avoid duplicates

_MESSAGE_CACHE_TTL = 60.0  # seconds


def _cleanup_cache():
    """Remove expired entries from dedup cache."""
    import time
    now = time.time()
    expired = [k for k, v in _processed_messages.items() if now - v > _MESSAGE_CACHE_TTL]
    for k in expired:
        del _processed_messages[k]


async def send_to_chatwoot(
    account_id: int,
    conversation_id: int,
    message: str,
) -> dict[str, Any]:
    """Send a message back to Chatwoot."""
    if not settings.chatwoot_url or not settings.chatwoot_token:
        return {"success": False, "error": "Chatwoot not configured"}

    url = f"{settings.chatwoot_url}/api/v1/accounts/{account_id}/conversations/{conversation_id}/messages"

    try:
        async with httpx.AsyncClient() as client:
            response = await client.post(
                url,
                headers={"api_access_token": settings.chatwoot_token},
                json={
                    "content": message,
                    "message_type": "outgoing",
                    "private": False,
                },
            )
            response.raise_for_status()
            data = response.json()
            return {"success": True, "message_id": data.get("id")}
    except Exception as e:
        logger.error(f"[Chatwoot] Send error: {e}")
        return {"success": False, "error": str(e)}


def should_process_webhook(payload: dict[str, Any]) -> tuple[bool, str]:
    """Validate webhook payload. Returns (should_process, reason)."""
    import time

    if payload.get("event") != "message_created":
        return False, f"Event type: {payload.get('event')}"

    if payload.get("message_type") != "incoming":
        return False, f"Message type: {payload.get('message_type')}"

    sender_type = (payload.get("sender") or {}).get("type")
    if sender_type in ("user", "agent_bot"):
        return False, f"Sender type: {sender_type}"

{% if webhooks.chatwoot.get("filter_open_conversations", true) %}
    conv_status = (payload.get("conversation") or {}).get("status")
    if conv_status == "open":
        return False, f"Conversation status: {conv_status} (human handling)"
{% endif %}

    msg_id = payload.get("id")
    if msg_id and msg_id in _processed_messages:
        return False, "Duplicate message"

    if msg_id:
        _processed_messages[msg_id] = time.time()
        _cleanup_cache()

    return True, "ok"


def extract_message_content(payload: dict[str, Any]) -> dict[str, Any]:
    """Extract text and attachments from payload."""
    return {
        "text": payload.get("content") or "",
        "attachments": payload.get("attachments") or [],
        "has_media": bool(payload.get("attachments")),

    inbox = payload.get("inbox") or {}

async def download_media(url: str, retries: int = 3) -> tuple[bytes, str] | None:
    """Download media from URL with retry."""
    for attempt in range(1, retries + 1):
        try:
            async with httpx.AsyncClient() as client:
                response = await client.get(url, follow_redirects=True, timeout=30.0)
                if response.status_code == 404 and attempt < retries:
                    import asyncio
                    await asyncio.sleep(attempt * 2)
                    continue
                response.raise_for_status()
                content_type = response.headers.get("content-type", "application/octet-stream")
                return response.content, content_type
        except Exception as e:
            if attempt < retries:
                import asyncio
                await asyncio.sleep(attempt * 2)
                continue
            logger.error(f"[Media] Download error: {e}")
            return None
    return None
